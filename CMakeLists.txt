cmake_minimum_required(VERSION 3.16)
project(roto_reaper VERSION 0.4.0 LANGUAGES CXX)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(reaper_roto MODULE src/extension.cpp src/config_ui.cpp vendor/WDL/sha.cpp)
set_target_properties(reaper_roto PROPERTIES PREFIX "")
if(APPLE)
 enable_language(OBJCXX)
 target_sources(reaper_roto PRIVATE vendor/WDL/swell/swell-modstub.mm)
 target_compile_definitions(reaper_roto PRIVATE SWELL_PROVIDED_BY_APP)
 target_link_libraries(reaper_roto PRIVATE "-framework Cocoa")
 set_target_properties(reaper_roto PROPERTIES SUFFIX ".dylib")
 target_link_options(reaper_roto PRIVATE "-undefined" "dynamic_lookup")
endif()
if(WIN32)
 enable_language(RC)
 target_sources(reaper_roto PRIVATE src/config_ui.rc)
 set_target_properties(reaper_roto PROPERTIES OUTPUT_NAME "reaper_roto" SUFFIX ".dll")
endif()
if(UNIX AND NOT APPLE)
 target_sources(reaper_roto PRIVATE vendor/WDL/swell/swell-modstub-generic.cpp)
 target_compile_definitions(reaper_roto PRIVATE SWELL_PROVIDED_BY_APP)
endif()
enable_testing()
add_executable(protocol_test tests/protocol_test.cpp vendor/WDL/sha.cpp)
add_test(NAME protocol COMMAND protocol_test)

add_executable(surface_test tests/surface_test.cpp vendor/WDL/sha.cpp)
add_test(NAME surface COMMAND surface_test)
add_test(NAME surface_ableton COMMAND surface_test ableton)
add_test(NAME surface_ableton_au COMMAND surface_test au)
add_test(NAME surface_ableton_vst2 COMMAND surface_test vst2)
if(APPLE)
 target_link_options(surface_test PRIVATE "-undefined" "dynamic_lookup")
endif()

foreach(target reaper_roto protocol_test surface_test)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4)
  target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wno-unused-parameter)
 endif()
endforeach()
